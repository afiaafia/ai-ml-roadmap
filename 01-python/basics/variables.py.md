# Variables in Python

# A variable stores a value that can be used later.
name = "Alex"
age = 25
country = "Canada"

print(name)
print(age)
print(country)


# Variables can store different types of values.
product = "Laptop"
quantity = 3
price = 799.99
is_available = True

print(product)
print(quantity)
print(price)
print(is_available)


# Multiple variables can be assigned in one line.
first_name, last_name, city = "John", "Smith", "Toronto"

print(first_name)
print(last_name)
print(city)


# The value of a variable can be changed.
score = 75
print(score)

score = 90
print(score)


# Variables can be used in calculations.
price = 120
quantity = 4
total_cost = price * quantity

print(total_cost)


# Variable names should be descriptive and follow Python naming rules.
student_name = "Alex"
total_marks = 450
is_passed = True

print(student_name)
print(total_marks)
print(is_passed)
