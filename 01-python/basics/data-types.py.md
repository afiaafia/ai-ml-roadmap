# Python Data Types

# Integer (int)
# Whole numbers without a decimal point.
quantity = 10
temperature = -5

print(quantity)
print(type(quantity))

print(temperature)
print(type(temperature))


# Floating-point number (float)
# Numbers that contain a decimal point.
price = 49.99
average_score = 87.5

print(price)
print(type(price))

print(average_score)
print(type(average_score))


# String (str)
# A sequence of characters enclosed in quotes.
product = "Notebook"
category = "Stationery"

print(product)
print(type(product))

print(category)
print(type(category))


# Boolean (bool)
# Represents either True or False.
is_available = True
is_completed = False

print(is_available)
print(type(is_available))

print(is_completed)
print(type(is_completed))


# List (list)
# An ordered and changeable collection of values.
numbers = [10, 20, 30, 40]

print(numbers)
print(type(numbers))


# Tuple (tuple)
# An ordered collection of values that cannot be changed.
coordinates = (10, 20)

print(coordinates)
print(type(coordinates))


# Dictionary (dict)
# Stores data as key-value pairs.
product_info = {
    "name": "Notebook",
    "price": 49.99,
    "quantity": 10
}

print(product_info)
print(type(product_info))


# Set (set)
# An unordered collection of unique values.
unique_numbers = {10, 20, 30, 10}

print(unique_numbers)
print(type(unique_numbers))


# NoneType
# None represents the absence of a value.
result = None

print(result)
print(type(result))


# Checking multiple data types
value_1 = 100
value_2 = 25.5
value_3 = "Python"
value_4 = True
value_5 = [1, 2, 3]

print(type(value_1))
print(type(value_2))
print(type(value_3))
print(type(value_4))
print(type(value_5))
